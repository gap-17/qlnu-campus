# 齐鲁师范学院章丘校区 3D 校园漫游 v0.1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build a runnable, no-install browser campus tour for QLNU Zhangqiu campus with massing, key facades, walk/fly/orbit modes, minimap, info cards and day/night, plus a Blender generator script.

**Architecture:** Static ES-module web app; Three.js loaded from a pinned CDN import map; all geometry generated at runtime from `buildings.json` and `campus.json`; pure math/data modules tested with Node's built-in test runner; a Blender Python script mirrors the same JSON to produce `.blend`/`.glb` when Blender is available.

**Tech Stack:** HTML/CSS, ES modules, Three.js 0.169 (CDN), Node 24 built-in test runner, Python 3.12 for the static server and Blender script.

**Spec:** `docs/specs/2026-11-22-campus-3d-design.md`

## Global Constraints

- Units: 1 = 1 m; origin at north gate; `+X` east, `+Z` south, `+Y` up.
- No npm dependencies and no build step; `web/` must run from `python -m http.server 8080 --directory web`.
- Every building has: `id, name, alias, category, status, floors, height_m, rect|polygon, entrance, description`.
- Building IDs are unique; object names in Blender are `bldg_<id>`.
- Chinese UI; names use “编号 + 常用别名”.
- Do not redistribute satellite/map tiles; they are tracing references only.
- Desktop first; mobile simplified. Target ≤ 250k triangles and ≤ 150 draw calls.

```js
export const CATEGORIES = [
  "gate", "teaching", "lab", "library", "public",
  "canteen", "dorm", "gym", "logistics"
];
export const STATUSES = ["built", "planned", "construction"];

export function buildingById(doc, id) {
  return doc.buildings.find((b) => b.id === id);
}

export function validateCampusData(buildingsDoc, campusDoc) {
  const errors = [];
  const seen = new Set();

  for (const b of buildingsDoc.buildings ?? []) {
    if (!b.id) errors.push("building missing id");
    if (seen.has(b.id)) errors.push(`duplicate id: ${b.id}`);
    seen.add(b.id);
    for (const key of ["name", "category", "status", "floors", "height_m", "entrance", "description"]) {
      if (b[key] === undefined) errors.push(`${b.id}: missing ${key}`);
    }
    if (!CATEGORIES.includes(b.category)) errors.push(`${b.id}: bad category ${b.category}`);
    if (!STATUSES.includes(b.status)) errors.push(`${b.id}: bad status ${b.status}`);
    if (!b.rect && !b.polygon) errors.push(`${b.id}: missing rect or polygon`);
    if (b.entrance && (typeof b.entrance.x !== "number" || typeof b.entrance.z !== "number")) {
      errors.push(`${b.id}: entrance must be {x,z}`);
    }
  }

  const gate = buildingById(buildingsDoc, "gate-north");
  if (!gate) errors.push("missing gate-north");
  else if (gate.entrance.x !== 0 || gate.entrance.z !== 0) {
    errors.push("gate-north entrance must be (0,0)");
  }

  if (!campusDoc.bounds) errors.push("campus missing bounds");
  if (!campusDoc.spawn) errors.push("campus missing spawn");
  return { ok: errors.length === 0, errors };
}
```

```js
export function rectToPolygon({ x, z, w, d, rot = 0 }) {
  const hw = w / 2, hd = d / 2;
  const a = (rot * Math.PI) / 180;
  const c = Math.cos(a), s = Math.sin(a);
  return [[-hw, -hd], [hw, -hd], [hw, hd], [-hw, hd]].map(([px, pz]) => [
    x + px * c - pz * s,
    z + px * s + pz * c
  ]);
}

export function pointInPolygon(x, z, polygon) {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const [xi, zi] = polygon[i], [xj, zj] = polygon[j];
    if ((zi > z) !== (zj > z) && x < ((xj - xi) * (z - zi)) / (zj - zi) + xi) {
      inside = !inside;
    }
  }
  return inside;
}

export function polygonArea(polygon) {
  let area = 0;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    area += polygon[j][0] * polygon[i][1] - polygon[i][0] * polygon[j][1];
  }
  return Math.abs(area / 2);
}

export function rectContains(rect, x, z) {
  const a = (-rect.rot * Math.PI) / 180;
  const dx = x - rect.x, dz = z - rect.z;
  const lx = dx * Math.cos(a) - dz * Math.sin(a);
  const lz = dx * Math.sin(a) + dz * Math.cos(a);
  return Math.abs(lx) <= rect.w / 2 && Math.abs(lz) <= rect.d / 2;
}
```

`collision.js`:

```js
export function resolveCircleAabb(pos, radius, box) {
  const nx = Math.max(box.minX, Math.min(pos.x, box.maxX));
  const nz = Math.max(box.minZ, Math.min(pos.z, box.maxZ));
  let dx = pos.x - nx, dz = pos.z - nz;
  const distSq = dx * dx + dz * dz;

  if (distSq > radius * radius) return { x: pos.x, z: pos.z, hit: false };
  if (distSq > 1e-9) {
    const dist = Math.sqrt(distSq);
    return { x: nx + (dx / dist) * radius, z: nz + (dz / dist) * radius, hit: true };
  }

  const left = pos.x - box.minX, right = box.maxX - pos.x;
  const up = pos.z - box.minZ, down = box.maxZ - pos.z;
  const min = Math.min(left, right, up, down);
  if (min === left) return { x: box.minX - radius, z: pos.z, hit: true };
  if (min === right) return { x: box.maxX + radius, z: pos.z, hit: true };
  if (min === up) return { x: pos.x, z: box.minZ - radius, hit: true };
  return { x: pos.x, z: box.maxZ + radius, hit: true };
}
```

```js
export function worldToMinimap(x, z, bounds, size) {
  return {
    u: ((x - bounds.minX) / (bounds.maxX - bounds.minX)) * size,
    v: ((z - bounds.minZ) / (bounds.maxZ - bounds.minZ)) * size
  };
}

export function minimapToWorld(u, v, bounds, size) {
  return {
    x: bounds.minX + (u / size) * (bounds.maxX - bounds.minX),
    z: bounds.minZ + (v / size) * (bounds.maxZ - bounds.minZ)
  };
}
```
